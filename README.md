function getExerciseById(id) {
  for (const d of Object.values(ROUTINE)) {
    const e = d.exercises.find(x => x.id === id);
    if (e) return e;
  }
  return MASTER_LIBRARY.find(x => x.id === id) || null;
}

function getDayExercises(dayKey) {
  const base = ROUTINE[dayKey].exercises;
  const custom = (data.customDays && data.customDays[dayKey]) || {added:[],removed:[]};
  let result = base.filter(ex => !custom.removed.includes(ex.id)).slice();
  for (const addedId of custom.added) {
    const ex = getExerciseById(addedId);
    if (ex && !result.find(e => e.id === ex.id)) result.push(ex);
  }
  // Apply custom ordering if present
  if (custom.order && custom.order.length) {
    const orderMap = new Map(custom.order.map((id,i)=>[id,i]));
    result.sort((a,b)=>{
      const ai = orderMap.has(a.id)?orderMap.get(a.id):999;
      const bi = orderMap.has(b.id)?orderMap.get(b.id):999;
      return ai-bi;
